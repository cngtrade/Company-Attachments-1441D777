# Website Putus Cinta

Website PHP untuk manage hubungan & proses putus.

## Fitur
- Register/Login user
- Request hubungan (add partner)
- End relationship (putus)
- History hubungan
- Dashboard sederhana

## Requirement
- PHP 8.0+
- MySQL 5.7+
- Apache/Nginx with mod_rewrite

## Setup

1. Import database:
```sql
mysql -u root -p < database.sql
```

2. Config database di `config/database.php`

3. Set document root ke folder project atau jalankan:
```
php -S localhost:8000
```

4. Akses: http://localhost:8000

```file name="database.sql"
CREATE DATABASE IF NOT EXISTS putus_cinta CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

USE putus_cinta;

CREATE TABLE users (
    id INT UNSIGNED PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    full_name VARCHAR(100) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_username (username),
    INDEX idx_email (email)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE relationships (
    id INT UNSIGNED PRIMARY KEY AUTO_INCREMENT,
    user_a_id INT UNSIGNED NOT NULL,
    user_b_id INT UNSIGNED NOT NULL,
    status ENUM('pending', 'active', 'ended', 'rejected') DEFAULT 'pending',
    requested_by INT UNSIGNED NOT NULL,
    started_at TIMESTAMP NULL,
    ended_at TIMESTAMP NULL,
    ended_by INT UNSIGNED NULL,
    end_reason TEXT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_users (user_a_id, user_b_id),
    INDEX idx_status (status),
    FOREIGN KEY (user_a_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (user_b_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (requested_by) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (ended_by) REFERENCES users(id) ON DELETE SET NULL,
    UNIQUE KEY unique_relationship (user_a_id, user_b_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE relationship_history (
    id INT UNSIGNED PRIMARY KEY AUTO_INCREMENT,
    relationship_id INT UNSIGNED NOT NULL,
    action ENUM('requested', 'accepted', 'rejected', 'ended') NOT NULL,
    performed_by INT UNSIGNED NOT NULL,
    reason TEXT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (relationship_id) REFERENCES relationships(id) ON DELETE CASCADE,
    FOREIGN KEY (performed_by) REFERENCES users(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```

```file name="config/database.php"
<?php
// Database configuration
define('DB_HOST', 'localhost');
define('DB_NAME', 'putus_cinta');
define('DB_USER', 'root');
define('DB_PASS', '');
define('DB_CHARSET', 'utf8mb4');

function getDBConnection() {
    try {
        $dsn = "mysql:host=" . DB_HOST . ";dbname=" . DB_NAME . ";charset=" . DB_CHARSET;
        $pdo = new PDO($dsn, DB_USER, DB_PASS, [
            PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
            PDO::ATTR_EMULATE_PREPARES => false
        ]);
        return $pdo;
    } catch (PDOException $e) {
        die("Connection failed: " . $e->getMessage());
    }
}

<?php
session_start();

function isLoggedIn() {
    return isset($_SESSION['user_id']);
}

function requireLogin() {
    if (!isLoggedIn()) {
        header('Location: login.php');
        exit;
    }
}

function getCurrentUserId() {
    return $_SESSION['user_id'] ?? null;
}

function getCurrentUsername() {
    return $_SESSION['username'] ?? null;
}

function loginUser($userId, $username) {
    $_SESSION['user_id'] = $userId;
    $_SESSION['username'] = $username;
    session_regenerate_id(true);
}

function logoutUser() {
    $_SESSION = array();
    if (isset($_COOKIE[session_name()])) {
        setcookie(session_name(), '', time() - 3600, '/');
    }
    session_destroy();
}

function generateCSRFToken() {
    if (!isset($_SESSION['csrf_token'])) {
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
    }
    return $_SESSION['csrf_token'];
}

function verifyCSRFToken($token) {
    return isset($_SESSION['csrf_token']) && hash_equals($_SESSION['csrf_token'], $token);
}

```file name="models/Relationship.php"
<?php
require_once __DIR__ . '/../config/database.php';

class Relationship {
    private $db;
    
    public function __construct() {
        $this->db = getDBConnection();
    }
    
    public function requestRelationship($requesterId, $partnerId) {
        // Check existing
        $stmt = $this->db->prepare("
            SELECT id, status FROM relationships 
            WHERE (user_a_id = ? AND user_b_id = ?) 
               OR (user_a_id = ? AND user_b_id = ?)
        ");
        $stmt->execute([$requesterId, $partnerId, $partnerId, $requesterId]);
        $existing = $stmt->fetch();
        
        if ($existing) {
            return ['success' => false, 'message' => 'Relationship already exists'];
        }
        
        // Create request
        $this->db->beginTransaction();
        try {
            $stmt = $this->db->prepare("
                INSERT INTO relationships (user_a_id, user_b_id, requested_by, status) 
                VALUES (?, ?, ?, 'pending')
            ");
            $stmt->execute([$requesterId, $partnerId, $requesterId]);
            $relationshipId = $this->db->lastInsertId();
            
            // Log history
            $stmt = $this->db->prepare("
                INSERT INTO relationship_history (relationship_id, action, performed_by) 
                VALUES (?, 'requested', ?)
            ");
            $stmt->execute([$relationshipId, $requesterId]);
            
            $this->db->commit();
            return ['success' => true, 'message' => 'Request sent'];
        } catch (Exception $e) {
            $this->db->rollBack();
            return ['success' => false, 'message' => 'Failed: ' . $e->getMessage()];
        }
    }
    
    public function acceptRelationship($relationshipId, $userId) {
        $stmt = $this->db->prepare("
            SELECT * FROM relationships 
            WHERE id = ? AND user_b_id = ? AND status = 'pending'
        ");
        $stmt->execute([$relationshipId, $userId]);
        $rel = $stmt->fetch();
        
        if (!$rel) {
            return ['success' => false, 'message' => 'Request not found'];
        }
        
        $this->db->beginTransaction();
        try {
            $stmt = $this->db->prepare("
                UPDATE relationships 
                SET status = 'active', started_at = NOW() 
                WHERE id = ?
            ");
            $stmt->execute([$relationshipId]);
            
            $stmt = $this->db->prepare("
                INSERT INTO relationship_history (relationship_id, action, performed_by) 
                VALUES (?, 'accepted', ?)
            ");
            $stmt->execute([$relationshipId, $userId]);
            
            $this->db->commit();
            return ['success' => true, 'message' => 'Relationship accepted'];
        } catch (Exception $e) {
            $this->db->rollBack();
            return ['success' => false, 'message' => 'Failed: ' . $e->getMessage()];
        }
    }
    
    public function endRelationship($relationshipId, $userId, $reason = '') {
        $stmt = $this->db->prepare("
            SELECT * FROM relationships 
            WHERE id = ? AND (user_a_id = ? OR user_b_id = ?) AND status = 'active'
        ");
        $stmt->execute([$relationshipId, $userId, $userId]);
        $rel = $stmt->fetch();
        
        if (!$rel) {
            return ['success' => false, 'message' => 'Active relationship not found'];
        }
        
        $this->db->beginTransaction();
        try {
            $stmt = $this->db->prepare("
                UPDATE relationships 
                SET status = 'ended', ended_at = NOW(), ended_by = ?, end_reason = ?
                WHERE id = ?
            ");
            $stmt->execute([$userId, $reason, $relationshipId]);
            
            $stmt = $this->db->prepare("
                INSERT INTO relationship_history (relationship_id, action, performed_by, reason) 
                VALUES (?, 'ended', ?, ?)
            ");
            $stmt->execute([$relationshipId, $userId, $reason]);
            
            $this->db->commit();
            return ['success' => true, 'message' => 'Relationship ended'];
        } catch (Exception $e) {
            $this->db->rollBack();
            return ['success' => false, 'message' => 'Failed: ' . $e->getMessage()];
        }
    }
    
    public function getActiveRelationships($userId) {
        $stmt = $this->db->prepare("
            SELECT r.*, 
                   u1.username as partner_username,
                   u1.full_name as partner_name
            FROM relationships r
            JOIN users u1 ON (u1.id = IF(r.user_a_id = ?, r.user_b_id, r.user_a_id))
            WHERE (r.user_a_id = ? OR r.user_b_id = ?) 
              AND r.status = 'active'
            ORDER BY r.started_at DESC
        ");
        $stmt->execute([$userId, $userId, $userId]);
        return $stmt->fetchAll();
    }
    
    public function getPendingRequests($userId) {
        $stmt = $this->db->prepare("
            SELECT r.*, 
                   u.username as requester_username,
                   u.full_name as requester_name
            FROM relationships r
            JOIN users u ON u.id = r.requested_by
            WHERE r.user_b_id = ? AND r.status = 'pending'
            ORDER BY r.created_at DESC
        ");
        $stmt->execute([$userId]);
        return $stmt->fetchAll();
    }
    
    public function getHistory($userId) {
        $stmt = $this->db->prepare("
            SELECT r.*, 
                   u1.username as partner_username,
                   u1.full_name as partner_name,
                   u2.username as ended_by_username
            FROM relationships r
            JOIN users u1 ON (u1.id = IF(r.user_a_id = ?, r.user_b_id, r.user_a_id))
            LEFT JOIN users u2 ON u2.id = r.ended_by
            WHERE (r.user_a_id = ? OR r.user_b_id = ?) 
              AND r.status = 'ended'
            ORDER BY r.ended_at DESC
        ");
        $stmt->execute([$userId, $userId, $userId]);
        return $stmt->fetchAll();
    }
}

    header('Location: dashboard.php');

$success = '';

    $username = trim($_POST['username'] ?? '');
    $email = trim($_POST['email'] ?? '');
    $password = $_POST['password'] ?? '';
    $confirm = $_POST['confirm_password'] ?? '';
    $fullName = trim($_POST['full_name'] ?? '');
    
    // Validasi
    if (empty($username) || empty($email) || empty($password) || empty($fullName)) {
        $error = 'Semua field harus diisi';
    } elseif (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
        $error = 'Email tidak valid';
    } elseif (strlen($username) < 3 || strlen($username) > 50) {
        $error = 'Username 3-50 karakter';
    } elseif (strlen($password) < 6) {
        $error = 'Password minimal 6 karakter';
    } elseif ($password !== $confirm) {
        $error = 'Password tidak cocok';
    } else {
        try {
            $db = getDBConnection();
            
            // Check existing
            $stmt = $db->prepare("SELECT id FROM users WHERE username = ? OR email = ?");
            $stmt->execute([$username, $email]);
            if ($stmt->fetch()) {
                $error = 'Username atau email sudah terdaftar';

                // Insert
                $hashedPassword = password_hash($password, PASSWORD_BCRYPT);
                $stmt = $db->prepare("
                    INSERT INTO users (username, email, password, full_name) 
                    VALUES (?, ?, ?, ?)
                ");
                $stmt->execute([$username, $email, $hashedPassword, $fullName]);
                
                $success = 'Registrasi berhasil! Silakan login.';
            }
        } catch (PDOException $e) {
            $error = 'Terjadi kesalahan: ' . $e->getMessage();

```file name="add-partner.php"
<?php
require_once 'includes/auth.php';
require_once 'config/database.php';
require_once 'models/Relationship.php';

requireLogin();

$error = '';
$success = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (!verifyCSRFToken($_POST['csrf_token'] ?? '')) {
        die('Invalid CSRF token');
    }
    
    $partnerUsername = trim($_POST['partner_username'] ?? '');
    
    if (empty($partnerUsername)) {
        $error = 'Username partner harus diisi';
    } else {
        $db = getDBConnection();
        $stmt = $db->prepare("SELECT id, username FROM users WHERE username = ?");
        $stmt->execute([$partnerUsername]);
        $partner = $stmt->fetch();
        
        if (!$partner) {
            $error = 'User tidak ditemukan';

            if ($result['success']) {

                $error = $result['message'];

}

    SELECT r.*, 
           u.username as partner_username,
           u.full_name as partner_name
    FROM relationships r
    JOIN users u ON (u.id = IF(r.user_a_id = ?, r.user_b_id, r.user_a_id))
    WHERE r.id = ? AND (r.user_a_id = ? OR r.user_b_id = ?) AND r.status = 'active'
